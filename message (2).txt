Table tipo_documento {
  id_tipo_documento int [pk]
  nombre varchar
}

Table personas {
  id_persona int [pk]
  nombre varchar
  apellido varchar
  id_tipo_documento int
  numero_documento varchar
  direccion varchar
  telefono varchar
  fecha_nacimiento date
  sexo varchar
}

Table rol {
  id_rol int [pk]
  nombre_rol varchar
  descripcion varchar
}

Table usuario {
  id_usuario int [pk]
  id_rol int
  email varchar
  password varchar
  activo boolean
  jwt_code text
}

Table cargo {
  id_cargo int [pk]
  nombre varchar
  codigo varchar
  descripcion varchar
  nivel_jerarquico int
}

Table empleado {
  id_empleado int [pk]
  id_persona int
  id_usuario int
  id_cargo int
  fecha_ingreso date
  fecha_retiro date
  activo boolean
}

Table paciente {
  id_paciente int [pk]
  id_persona int
  activo boolean
}

Table estado_cita {
  id_estado_cita int [pk]
  codigo varchar
  descripcion varchar
}

Table cita {
  id_cita int [pk]
  id_paciente int
  id_medico int
  id_estado_cita int
  fecha date
  hora_inicio time
  hora_fin time
  motivo_consulta varchar
  observaciones text
  id_usuario_creacion int
  fecha_creacion datetime
}

Table detalle_cita {
  id_detalle int [pk]
  id_cita int
  id_profesional int
  nota_atencion text
  resumen_consulta text
  fecha_registro datetime
}

Table especialidad {
  id_especialidad int [pk]
  nombre varchar
  descripcion varchar
}

Table medico {
  id_medico int [pk]
  id_empleado int
  id_especialidad int
}

Table disponibilidad {
  id_disponibilidad int [pk]
  id_medico int
  dia_semana int
  hora_inicio time
  hora_fin time
  activo boolean
  fecha_inicio_vigencia date
  fecha_fin_vigencia date
}

Table bloque_horario {
  id_bloque int [pk]
  id_disponibilidad int
  fecha date
  hora_inicio time
  hora_fin time
  ocupado boolean
}

Table diagnostico {
  id_diagnostico int [pk]
  codigo_cie10 varchar
  descripcion varchar
  activo boolean
}

Table detalle_diagnostico {
  id_detalle_diagnostico int [pk]
  id_detalle_cita int
  id_diagnostico int
  principal boolean
}

Table tratamiento {
  id_tratamiento int [pk]
  codigo varchar
  nombre varchar
  descripcion varchar
  dosis varchar
  frecuencia varchar
  duracion_dias int
  indicaciones text
  activo boolean
}

Table atencion_tratamiento {
  id_atencion_tratamiento int [pk]
  id_detalle_cita int
  id_tratamiento int
  dosis_personalizada varchar
  frecuencia_personalizada varchar
  duracion_dias_personalizada int
  indicaciones_personalizadas text
}
